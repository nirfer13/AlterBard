# Bot dependencies. Versions match pyproject.toml / poetry.lock.
# Install with: pip install -r requirements.txt
#
# Playback is handled by the Lavalink server (Lavalink.jar + application.yml),
# which plays files from music/. yt-dlp searches and downloads them (see
# music_library.py); imageio-ffmpeg only supplies an ffmpeg binary for it.
#
# yt-dlp is a floor, not a pin: YouTube breaks old versions every few weeks.
# When downloads start failing, the first thing to try is
#     pip install -U "yt-dlp[default]"

# wavelink must be 3.5.1+ : Lavalink 4.2 requires the "channelId" field in the
# voice state payload, and older wavelink does not send it - the player is then
# rejected and destroyed the moment it connects. 3.5.1 also brought DAVE support.

wavelink==3.5.2
python-dotenv==1.0.0
yarl==1.9.2
yt-dlp[default]>=2026.10.21
